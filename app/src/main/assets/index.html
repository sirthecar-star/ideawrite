💡 Supabase 테이블 및 RLS 권한 SQL

Supabase SQL Editor에서 아래 코드를 실행하면 저장/수정/삭제 권한 오류가 100% 해결됩니다: 

-- 1. 테이블 생성 (모든 컬럼 및 인덱스)
create table if not exists public.ideas (
  id text primary key,
  title text not null default '',
  content text default '',
  category text default '일상',
  importance integer default 3,
  tags text[] default '{}',
  is_pinned boolean default false,
  is_favorite boolean default false,
  color_hex text default '#ffffff',
  voice_audio text default null,
  created_at bigint not null default (extract(epoch from now()) * 1000)::bigint,
  updated_at bigint not null default (extract(epoch from now()) * 1000)::bigint
);

-- 2. RLS 정책 전체 허용 (익명/웹 접근 권한 부여)
alter table public.ideas enable row level security;
drop policy if exists "Enable all access for all users" on public.ideas;
drop policy if exists "Public Access" on public.ideas;
create policy "Enable all access for all users" 
on public.ideas for all to public using (true) with check (true);
